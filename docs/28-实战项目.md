# 28-实战项目：智能旅行管家

> 📅 适用于 LangGraph.js v1.x | 👶 零基础友好
> 💡 **注意：** 本章综合运用 LangGraph 篇全部知识，是毕业考试！

---

## 🎯 项目目标

把 LangChain 教程第 12 章的"旅行助手"用 LangGraph 重新打造，升级成真正可用的形态：

| 能力 | 用到的知识 | 章节 |
|------|-----------|------|
| 工具调用（查天气/酒店/预订） | ReAct Agent | 第 20 章 |
| 多轮对话记忆 | MemorySaver + thread_id | 第 21 章 |
| **下单前人工确认** | interrupt + Command resume | 第 22 章 |
| 打字机实时输出 | stream + messages 模式 | 第 25 章 |
| 自定义进度上报 | config.writer + custom 模式 | 第 25 章 |

## 🧠 设计思路

```
用户: "帮我查三亚天气，订一家酒店"
        │
        ▼
┌─────────────────────────────────────────────┐
│  ReAct Agent（createReactAgent）             │
│                                             │
│  工具: get_weather   查天气                  │
│        search_hotel  查酒店（带进度上报）      │
│        book_hotel    预订酒店（★ 下单前 interrupt 人工确认）│
│                                             │
│  记忆: MemorySaver + thread_id（多轮对话）    │
└─────────────────────────────────────────────┘
        │
        ▼
  stream(messages) 打字机输出给用户
        │
        ▼
book_hotel 触发 interrupt → 用户回复"确认/取消" → Command({ resume }) 恢复
```

## 💡 完整代码

创建 `travel-agent.js`：

```javascript
import "dotenv/config";
import { ChatOpenAI } from "@langchain/openai";
import { HumanMessage } from "@langchain/core/messages";
import { tool } from "@langchain/core/tools";
import { createReactAgent } from "@langchain/langgraph/prebuilt";
import { MemorySaver, interrupt, Command } from "@langchain/langgraph";
import { z } from "zod";

// ===== 1. 模型 =====
const model = new ChatOpenAI({
  modelName: "MiniMax/MiniMax-M2.5",
  apiKey: process.env.MODELSCOPE_API_KEY,
  temperature: 0,
  configuration: {
    baseURL: "https://api-inference.modelscope.cn/v1",
  },
});

// ===== 2. 工具集 =====

// 工具 1：查天气（模拟数据）
const weatherTool = tool(
  async ({ city }) => {
    const data = {
      "三亚": "晴，28°C，适合海边活动",
      "北京": "多云，18°C，早晚温差大",
      "成都": "小雨，22°C，记得带伞",
    };
    return data[city] || `${city}：晴，25°C`;
  },
  {
    name: "get_weather",
    description: "查询城市天气，用于判断是否适合出行",
    schema: z.object({ city: z.string().describe("城市名称") }),
  }
);

// 工具 2：查酒店（用 writer 上报进度）
const searchHotelTool = tool(
  async ({ city }, config) => {
    // 第二个参数 config 里有 writer，可向流式输出发送进度
    config?.writer?.({ type: "progress", message: `正在检索 ${city} 的酒店...` });
    await new Promise((r) => setTimeout(r, 500));  // 模拟网络延迟
    config?.writer?.({ type: "progress", message: "比价完成" });

    return `${city} 推荐酒店：\n` +
      `1. 海景湾酒店 - 680 元/晚（海边 500 米）\n` +
      `2. 舒适快捷酒店 - 320 元/晚（市中心）`;
  },
  {
    name: "search_hotel",
    description: "查询指定城市的酒店推荐及价格",
    schema: z.object({
      city: z.string().describe("城市名称"),
    }),
  }
);

// 工具 3：预订酒店（★ 下单前人工确认）
const bookHotelTool = tool(
  async ({ hotel, nights }) => {
    // 暂停执行，等人工确认 —— interrupt 前不要放副作用！
    const decision = interrupt({
      action: "book_hotel",
      hotel,
      nights,
      message: `确认预订 ${hotel} ${nights} 晚吗？`,
    });

    // 恢复后 decision 就是用户的确认结果
    if (decision?.approved) {
      return `预订成功！${hotel} ${nights} 晚，确认号 TR-2026-8899`;
    }
    return "用户取消了预订";
  },
  {
    name: "book_hotel",
    description: "预订酒店，需要提供酒店名称和入住晚数（预订前会请求用户确认）",
    schema: z.object({
      hotel: z.string().describe("酒店名称"),
      nights: z.number().describe("入住晚数"),
    }),
  }
);

// ===== 3. 创建带记忆的 Agent =====
const agent = createReactAgent({
  llm: model,
  tools: [weatherTool, searchHotelTool, bookHotelTool],
  prompt: "你是一位专业的旅行管家，帮用户查询天气、推荐酒店。涉及花钱的操作必须先调用工具让用户确认。",
  checkpointSaver: new MemorySaver(),   // 多轮对话记忆
});

// ===== 4. 封装一个"对话回合"：流式输出 + 中断处理 =====
async function chat(threadId, userMessage) {
  const config = { configurable: { thread_id: threadId } };

  // 同时订阅 token 流和进度上报
  const stream = await agent.stream(
    { messages: [new HumanMessage(userMessage)] },
    { streamMode: ["messages", "custom"], ...config }
  );

  let interrupted = null;

  for await (const [mode, data] of stream) {
    if (mode === "messages") {
      const [chunk, metadata] = data;
      // 只输出最终回答节点的 token（过滤工具内部模型调用）
      if (chunk.content && typeof chunk.content === "string") {
        process.stdout.write(chunk.content);
      }
    } else if (mode === "custom") {
      console.log(`\n  ⚙️ [进度] ${data.message}`);
    }
  }

  // 检查是否发生了 interrupt：从最新的状态快照里读任务的中断信息
  const state = await agent.getState(config);
  if (state.tasks.length > 0 && state.tasks[0].interrupts?.length > 0) {
    interrupted = state.tasks[0].interrupts[0].value;
  }
  return interrupted;
}

// ===== 5. 运行整个流程 =====
const THREAD = "trip-2026";

// --- 第一轮：查询（流式打字机输出）---
console.log("👤 用户: 帮我看看三亚天气怎么样，顺便推荐个酒店");
console.log("🤖 管家: ");
const interruptPayload = await chat(THREAD, "帮我看看三亚天气怎么样，顺便推荐个酒店");

// --- 第二轮：下单（会触发人工确认）---
if (!interruptPayload) {
  console.log("\n\n👤 用户: 就订海景湾酒店，住 3 晚");
  console.log("🤖 管家: ");
  const payload = await chat(THREAD, "就订海景湾酒店，住 3 晚");

  if (payload) {
    // ===== 图暂停了，等待人工确认 =====
    console.log("\n\n⏸️ 需要确认:", JSON.stringify(payload, null, 2));

    // 真实场景：在这里弹窗/发消息给用户，等待点击
    const userSaysYes = true;  // 模拟用户点击"确认"

    console.log(`\n👤 用户: ${userSaysYes ? "确认预订" : "不订了"}`);
    console.log("🤖 管家: ");
    await resume(THREAD, { approved: userSaysYes });
  }
}

// 恢复执行的封装
async function resume(threadId, resumeValue) {
  const config = { configurable: { thread_id: threadId } };

  const stream = await agent.stream(
    new Command({ resume: resumeValue }),   // ★ 恢复输入
    { streamMode: ["messages", "custom"], ...config }
  );

  for await (const [mode, data] of stream) {
    if (mode === "messages") {
      const [chunk] = data;
      if (chunk.content && typeof chunk.content === "string") {
        process.stdout.write(chunk.content);
      }
    }
  }
}
```

> 💡 **注意：** `streamMode` 传数组时，`config`（含 `configurable`）需要与模式一起放进第二个参数——所以上面用 `{ streamMode: [...], ...config }` 展开。

### 这个项目里藏着所有知识点

| 你看到的 | 背后的 LangGraph 机制 |
|---------|---------------------|
| `⚙️ [进度]` 实时弹出 | `config.writer` + `custom` 流模式 |
| 回答逐字出现 | `messages` 流模式 |
| 第二轮还记得三亚的上下文 | `MemorySaver` + `thread_id` |
| 下单前停下来等确认 | 工具内 `interrupt` + 检查点 |
| "确认预订"后继续跑完 | `new Command({ resume })` |

## 🚀 进阶方向（留给你的作业）

1. **加时间旅行调试**：用 `getStateHistory` 打印 Agent 每一步的状态，观察它思考了什么
2. **换持久化**：把 `MemorySaver` 换成 SQLite Checkpointer，重启进程后记忆还在
3. **加长期记忆**：用 `InMemoryStore`（第 26 章）记住用户的旅行偏好，下次对话直接推荐
4. **拆专家**：把"查天气"和"订酒店"拆成两个子 Agent，用 `createSupervisor` 组队（第 24 章）
5. **接前端**：用 WebSocket 把 `stream` 的三种数据（token/进度/中断请求）推给网页，做一个真正的聊天界面
6. **试试 Functional API**：用 `entrypoint` + `task`（第 27 章）改写审批流程，对比两种风格
